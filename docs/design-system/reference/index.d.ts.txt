import type * as React from 'react';

export type IconName = 'copy' | 'check' | 'refresh' | 'loader' | 'close' | 'chevron-down' | 'upload' | 'file' | 'pencil' | 'sparkle' | 'sparkles' | 'alert';
export interface IconProps { name: IconName; className?: string }
export declare function Icon(props: IconProps): React.ReactElement;

export interface ButtonProps extends React.ButtonHTMLAttributes<HTMLButtonElement> {
  variant?: 'primary' | 'secondary' | 'ghost';
  size?: 'lg' | 'md' | 'sm';
  /** Shows the spinner, keeps the label for screen readers, blocks clicks via aria-busy styling. */
  loading?: boolean;
  icon?: IconName;
}
export declare function Button(props: ButtonProps): React.ReactElement;

export interface IconButtonProps extends React.ButtonHTMLAttributes<HTMLButtonElement> {
  icon: IconName;
  /** Required: becomes aria-label and the tooltip. */
  label: string;
  size?: 'md' | 'sm';
}
export declare function IconButton(props: IconButtonProps): React.ReactElement;

export interface TextFieldProps {
  label: string;
  value?: string;
  defaultValue?: string;
  onChange?: (e: React.ChangeEvent<HTMLInputElement | HTMLTextAreaElement>) => void;
  placeholder?: string;
  helper?: string;
  /** Replaces the helper text and turns the field red. */
  error?: string;
  /** Shows the "n / max" counter and caps input. */
  maxLength?: number;
  disabled?: boolean;
  id?: string;
  autoFocus?: boolean;
}
export declare function TextField(props: TextFieldProps): React.ReactElement;
export interface TextAreaProps extends TextFieldProps { rows?: number }
export declare function TextArea(props: TextAreaProps): React.ReactElement;

export interface ChipProps { selected?: boolean; disabled?: boolean; onClick?: () => void; children?: React.ReactNode }
export declare function Chip(props: ChipProps): React.ReactElement;

export interface ChipOption { value: string; label: string; disabled?: boolean }
export interface ChipGroupProps {
  label: string;
  options: ChipOption[];
  value?: string;
  defaultValue?: string;
  onChange?: (value: string) => void;
  disabled?: boolean;
}
export declare function ChipGroup(props: ChipGroupProps): React.ReactElement;

export interface RefineOption { value: string; label: string; icon?: IconName }
export interface ResultPart { label: string; text: string }
export interface ResultCardProps {
  label: string;
  text: string;
  /** Multi-part copy, e.g. Title / Body / Button for an error. Shown as labelled lines; Copy copies all parts, one per paragraph. Hides the character count. */
  parts?: ResultPart[];
  /** Small blue pill after the label, e.g. "Recommended". */
  badge?: string;
  /** One line under the copy explaining what this option does differently, e.g. "Names the problem and the next step". */
  rationale?: string;
  onCopy?: (text: string) => void;
  /** Optional quick actions under the text ("Shorter", "More casual", "Try again"). Omit to show none. */
  refineOptions?: RefineOption[];
  /** Called with the option's value, or with "custom" and the person's own instruction when allowCustom is on. */
  onRefine?: (value: string, customText?: string) => void;
  /** Adds an "Other" pill that opens a one-line field where the person writes what should change. */
  allowCustom?: boolean;
  customPlaceholder?: string;
  /** Swaps the text for a loading placeholder while this one option is being refined. */
  refining?: boolean;
  /** Short note after the label once refined, e.g. "Shorter". */
  note?: string;
  /** Shows "n / max characters" under the text, in red with the overage when the text is too long. */
  maxChars?: number;
  /** Shows an Undo action that restores the previous text. */
  onUndo?: () => void;
}
export declare function ResultCard(props: ResultCardProps): React.ReactElement;

export interface ResultSkeletonProps { lines?: number }
export declare function ResultSkeleton(props: ResultSkeletonProps): React.ReactElement;

export interface ToastProps { icon?: IconName; children?: React.ReactNode }
export declare function Toast(props: ToastProps): React.ReactElement;

export interface EmptyStateProps { title: string; icon?: IconName; action?: React.ReactNode; children?: React.ReactNode }
export declare function EmptyState(props: EmptyStateProps): React.ReactElement;

export interface DropdownOption { value: string; label: string; disabled?: boolean }
export interface DropdownProps {
  label: string;
  options: DropdownOption[];
  value?: string;
  defaultValue?: string;
  onChange?: (e: React.ChangeEvent<HTMLSelectElement>) => void;
  /** Shown in grey until something is picked; can't be re-selected. */
  placeholder?: string;
  helper?: string;
  /** Replaces the helper text and turns the field red. */
  error?: string;
  disabled?: boolean;
  id?: string;
}
export declare function Dropdown(props: DropdownProps): React.ReactElement;

export interface UploadedFile { name: string; size: number; type: string; url: string }
export interface UploaderProps {
  label?: string;
  /** Second line in the zone, e.g. "JPG, PNG or GIF up to 5 MB". */
  hint?: string;
  /** Main line in the zone. Default "Upload or drop file here". */
  cta?: string;
  /** Same format as the input's accept attribute, e.g. "image/png, image/jpeg, .gif". Checked on drop too. */
  accept?: string;
  /** Largest file in bytes. Bigger files show an error instead of being taken. */
  maxSize?: number;
  /** Called with the file and, for images, a data URL for previews. */
  onFile?: (file: File, dataUrl: string) => void;
  onRemove?: () => void;
  /** Control which file is shown; omit to let the uploader keep it. */
  value?: UploadedFile | null;
  error?: string;
  typeError?: string;
  disabled?: boolean;
  id?: string;
}
export declare function Uploader(props: UploaderProps): React.ReactElement;

export interface LogoProps {
  /** "lockup" = mark + wordmark (default); "mark" = the square alone, labelled "Copymopy" for screen readers. */
  variant?: 'lockup' | 'mark';
  /** Mark size in px (default 32). The wordmark scales with it. At 20px and below the mark uses the two-pill cut. */
  size?: number;
  /** White mark and wordmark, for blue-main or dark grounds. */
  inverse?: boolean;
}
export declare function Logo(props: LogoProps): React.ReactElement;

declare global {
  interface Window {
    Copymopy: {
      Button: typeof Button; IconButton: typeof IconButton; TextField: typeof TextField; TextArea: typeof TextArea;
      Chip: typeof Chip; ChipGroup: typeof ChipGroup; ResultCard: typeof ResultCard; ResultSkeleton: typeof ResultSkeleton;
      Toast: typeof Toast; EmptyState: typeof EmptyState; Icon: typeof Icon; Logo: typeof Logo; Dropdown: typeof Dropdown; Uploader: typeof Uploader;
    };
  }
}
